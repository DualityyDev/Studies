int main()
{
	SetConsoleCP(1252);
	SetConsoleOutputCP(1252);

	double valor;
	
	conta.iniciarConta();
	do{
		cout << endl

		<< "1 - Consultar saldo" << endl
		<< "2 - Depositar dinheiro" << endl
		<< "3 - Levantar dinheiro" << endl
		<< "4 - Sair" << endl

					conta.depositar(valor);

				else

			case 3:

					}

// .h
#ifndef CONTA_H
#define CONTA_H

class Conta
{
    private:
        double saldo;
    public:
        void iniciarConta();
        void depositar(double valor);
        bool levantar(double valor);
        double consultarSaldo();
};

#endif
// .cpp
#include "Conta.h"

void Conta::iniciarConta()
{
    saldo = 500.0;
}

void Conta::depositar(double valor)
{
    if (valor > 0)
        saldo = saldo + valor;
}

bool Conta::levantar(double valor)
{
    if (valor > 0 && valor <= saldo)
    {
        saldo = saldo - valor;
        return true;
    }
    else
    {
        return false;
    }
}

double Conta::consultarSaldo()
{
    return saldo;
}
